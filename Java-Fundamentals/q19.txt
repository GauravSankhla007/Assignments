/*"Write a program to calculate sum of series for N = 5, it should work for any value of N.
1 - 2/!2 + 3/!3 - 4/!4 + 5/!5"
 */
import java.util.Scanner;
public class q19_1 {
    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);
        System.out.print("Enter a number: ");
        int N = scanner.nextInt();
        double sum = 0;
        for(int i = 1;i<=N;i++){
            double fact = 1;
            for(int j = i;j>=1;j--){
                fact = fact * j ;
            }
            double term = i/fact;
            if(i%2==0){
                sum = sum -term;
            }
            else{
                sum = sum+term;
            }
        }
        System.out.println(sum);
    }
}
