import java.util.Scanner;
public class q15_1 {
    public static void main(String[] args) {
        for (int i = 1; i <= 5; i++) {
            if (i <= 3) {
                for (int j = 1; j <= 4 - i; j++) {
                    System.out.print(" ");
                }
                for (int j = i; j >= 1; j--) {
                    System.out.print(j);
                }
                for (int j = 2; j <= i; j++) {
                    System.out.print(j);
                }
            } else {
                for (int j = 1; j <= i - 2; j++) {
                    System.out.print(" ");
                }
                for (int k = 6 - i; k >= 1; k--) {
                    System.out.print(k);
                }
                for (int k = 2; k <= 6 - i; k++) {
                    System.out.print(k);
                }
            }
            System.out.println("");

        }
    }
}
