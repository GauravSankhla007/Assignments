/*"Print a Design like this
   1
  121
 12321
  121
   1"
 */
public class q16_1 {
    public static void main(String[] args) {

        for(int i=1;i<=5;i++){
            if(i<=3) {
                for (int j = 1; j <= 4 - i; j++) {
                    System.out.print(" ");
                }
                for(int j = 1;j<=i;j++){
                    System.out.print(j);
                }
                for(int j = i-1;j>=1;j--){
                    System.out.print(j);
                }
            }
            else {
                int start = 6-i;

                for(int j = 1;j<=i-2;j++){
                    System.out.print(" ");
                }
                for (int j =1;j<=start;j++){
                    System.out.print(j);
                }
                for(int j = start-1;j>=1;j--){
                    System.out.print(j);
                }
            }
            System.out.println("");
        }
    }
}
