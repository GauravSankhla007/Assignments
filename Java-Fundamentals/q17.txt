/*"Print a Design like this
32123
21 12
1   1"
*/
public class q17_1 {
    public static void main(String[] args) {


        for(int i=1; i<=3;i++){
            for(int j=4-i;j>=1;j--){
                System.out.print(j);
            }
            for (int j = 1; j <= (i - 1) * i / 2; j++) {
                System.out.print(" ");
            }
            if (i==1){
                for(int j = 2;j<=3;j++){
                    System.out.print(j);
                }
            }
            else{
                for(int j = 1;j<=4-i;j++){
                    System.out.print(j);
                }
            }


            System.out.println("");
        }

    }
}
