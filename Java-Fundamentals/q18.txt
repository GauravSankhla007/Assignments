/*"Print a Design like this
1     1
12   21
123 321
1234321"
 */
public class q18_1 {
    public static void main(String[] args) {

        for(int i = 1;i<=4;i++){

            for(int j=1;j<=i;j++){
                System.out.print(j);
            }
            for(int j = 7-2*i;j>=1;j--){
                System.out.print(" ");
            }
            if(i<=3){
            for(int j = i;j>=1;j--) {
                System.out.print(j);
            }
            }
            else {
                for(int j = i-1;j>=1;j--){
                    System.out.print(j);
                }
            }
            System.out.println("");
        }

    }
}
